//go:build darwin || linux

package client

import (
	"crypto/rand"
	"fmt"
	"testing"
	"time"
)

// BenchmarkCatalogIdle measures real production refresh timers and no application
// streams. Use -benchtime=1x: each operation observes a 22-second idle interval.
func BenchmarkCatalogIdle(b *testing.B) {
	for _, devices := range []int{2, 5, 20} {
		for _, services := range []int{1, 25, 100} {
			b.Run(fmt.Sprintf("devices%d/services%d", devices, services), func(b *testing.B) {
				info, token, httpClient := benchmarkRelay(b, false)
				group := make([]byte, 32)
				rand.Read(group)
				clients := make([]*Client, devices)
				registries := make([]*httpBenchRegistry, devices)
				for i := range clients {
					clients[i], registries[i] = benchmarkClient(b, info.URL, token, group, httpClient, "", services)
				}
				until := time.Now().Add(10 * time.Second)
				for {
					complete := true
					for _, c := range clients {
						s := c.Snapshot()
						if !s.Connected || len(s.Records) != devices {
							complete = false
						}
					}
					if complete {
						break
					}
					if time.Now().After(until) {
						b.Fatal("catalog convergence timeout")
					}
					time.Sleep(20 * time.Millisecond)
				}
				time.Sleep(500 * time.Millisecond)
				sequences := make([]uint64, devices)
				for i, r := range registries {
					r.mu.Lock()
					sequences[i] = r.record.Sequence
					r.mu.Unlock()
				}
				before := relayMetrics(b, info.Metrics)
				b.ResetTimer()
				start := time.Now()
				time.Sleep(time.Duration(b.N) * 22 * time.Second)
				elapsed := time.Since(start).Seconds()
				b.StopTimer()
				after := relayMetrics(b, info.Metrics)
				publications := uint64(0)
				for i, c := range clients {
					s := c.Snapshot()
					if !s.Connected || s.Error != "" || len(s.Records) != devices {
						b.Fatalf("unhealthy catalog: connected=%v records=%d error=%s", s.Connected, len(s.Records), s.Error)
					}
					r := registries[i]
					r.mu.Lock()
					publications += r.record.Sequence - sequences[i]
					r.mu.Unlock()
				}
				if publications == 0 {
					b.Fatal("no periodic publication observed")
				}
				// Report observed rates; the window contains a burst, not uniformly spaced work.
				b.ReportMetric(float64(after.In-before.In)*60/elapsed, "relay-in-B/min")
				b.ReportMetric(float64(after.Out-before.Out)*60/elapsed, "relay-out-B/min")
				b.ReportMetric((after.CPU-before.CPU)*60/elapsed, "relay-CPU-s/min")
				b.ReportMetric(float64(after.AllocBytes-before.AllocBytes)*60/elapsed, "relay-alloc-B/min")
				b.ReportMetric(float64(after.Allocations-before.Allocations)*60/elapsed, "relay-allocs/min")
				b.ReportMetric(float64(after.Heap)/(1<<20), "relay-heap-MiB")
				b.ReportMetric(float64(publications), "publications")
			})
		}
	}
}
